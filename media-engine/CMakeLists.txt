cmake_minimum_required(VERSION 3.20)
project(streamforge_media LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.22 gstreamer-webrtc-1.0 gstreamer-sdp-1.0)
pkg_check_modules(SOUP REQUIRED IMPORTED_TARGET libsoup-3.0 json-glib-1.0)
add_executable(streamforge-media src/main.cpp)
target_link_libraries(streamforge-media PRIVATE PkgConfig::GST PkgConfig::SOUP)
target_compile_definitions(streamforge-media PRIVATE GST_USE_UNSTABLE_API)
target_compile_options(streamforge-media PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS streamforge-media DESTINATION bin)
